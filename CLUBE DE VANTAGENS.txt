    <script>
        // ⚠️ IMPORTANTE: Substitua 'SEU_FORM_ID_AQUI' pelo seu Form ID do Formspree
        // Exemplo: 'f/a1b2c3d4e5f6'
        const FORMSPREE_FORM_ID =f/xpwopbjq;

        // Inicializar Canvas de Assinatura
        const canvas = document.getElementById('signatureCanvas');
        const ctx = canvas.getContext('2d');
        let isDrawing = false;
        let hasSignature = false;

        // Ajustar tamanho do canvas
        function resizeCanvas() {
            const rect = canvas.getBoundingClientRect();
            canvas.width = rect.width;
            canvas.height = rect.height;
        }

        resizeCanvas();
        window.addEventListener('resize', resizeCanvas);

        // Eventos de desenho
        canvas.addEventListener('mousedown', (e) => {
            isDrawing = true;
            const rect = canvas.getBoundingClientRect();
            const x = e.clientX - rect.left;
            const y = e.clientY - rect.top;
            ctx.beginPath();
            ctx.moveTo(x, y);
        });

        canvas.addEventListener('mousemove', (e) => {
            if (!isDrawing) return;
            const rect = canvas.getBoundingClientRect();
            const x = e.clientX - rect.left;
            const y = e.clientY - rect.top;
            ctx.lineTo(x, y);
            ctx.strokeStyle = '#333';
            ctx.lineWidth = 2;
            ctx.lineCap = 'round';
            ctx.lineJoin = 'round';
            ctx.stroke();
            hasSignature = true;
        });

        canvas.addEventListener('mouseup', () => {
            isDrawing = false;
        });

        canvas.addEventListener('mouseleave', () => {
            isDrawing = false;
        });

        // Suporte para touch (mobile)
        canvas.addEventListener('touchstart', (e) => {
            e.preventDefault();
            isDrawing = true;
            const rect = canvas.getBoundingClientRect();
            const touch = e.touches[0];
            const x = touch.clientX - rect.left;
            const y = touch.clientY - rect.top;
            ctx.beginPath();
            ctx.moveTo(x, y);
        });

        canvas.addEventListener('touchmove', (e) => {
            e.preventDefault();
            if (!isDrawing) return;
            const rect = canvas.getBoundingClientRect();
            const touch = e.touches[0];
            const x = touch.clientX - rect.left;
            const y = touch.clientY - rect.top;
            ctx.lineTo(x, y);
            ctx.strokeStyle = '#333';
            ctx.lineWidth = 2;
            ctx.lineCap = 'round';
            ctx.lineJoin = 'round';
            ctx.stroke();
            hasSignature = true;
        });

        canvas.addEventListener('touchend', () => {
            isDrawing = false;
        });

        // Limpar assinatura
        function clearSignature() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            hasSignature = false;
        }

        // Alternar entre abas
        function switchTab(tabName) {
            const contents = document.querySelectorAll('.tab-content');
            const buttons = document.querySelectorAll('.tab-button');

            contents.forEach(content => content.classList.remove('active'));
            buttons.forEach(button => button.classList.remove('active'));

            document.getElementById(tabName).classList.add('active');
            event.target.classList.add('active');
        }

        // Salvar dados e ir para visualização
        function saveAndPreview() {
            const form = document.getElementById('declarationForm');
            
            // Validar formulário
            if (!form.checkValidity()) {
                showError('Por favor, preencha todos os campos obrigatórios.');
                return;
            }

            // Validar assinatura
            if (!hasSignature) {
                showError('Por favor, assine a declaração.');
                return;
            }

            // Obter valores
            const nomeCompleto = document.getElementById('nomeCompleto').value;
            const nomeEmpresa = document.getElementById('nomeEmpresa').value;
            const cnpj = document.getElementById('cnpj').value;
            const beneficio = document.getElementById('beneficio').value;
            const local = document.getElementById('local').value;
            const data = document.getElementById('data').value;
            const signatureImage = canvas.toDataURL('image/png');

            // Atualizar visualização
            document.getElementById('preview-nome').textContent = nomeCompleto;
            document.getElementById('preview-empresa').textContent = nomeEmpresa;
            document.getElementById('preview-cnpj').textContent = cnpj;
            document.getElementById('preview-beneficio').textContent = beneficio;
            document.getElementById('preview-local').textContent = local + ', ' + data;

            // Mostrar assinatura na visualização
            const signatureImg1 = document.getElementById('signatureImagePreview');
            const signatureImg2 = document.getElementById('signatureImagePreview2');
            signatureImg1.src = signatureImage;
            signatureImg1.style.display = 'block';
            signatureImg2.src = signatureImage;
            signatureImg2.style.display = 'block';

            // Salvar dados no localStorage para envio posterior
            localStorage.setItem('declaracaoData', JSON.stringify({
                nomeCompleto,
                nomeEmpresa,
                cnpj,
                beneficio,
                local,
                data,
                email: document.getElementById('email').value,
                signatureImage
            }));

            // Ir para aba de visualização
            document.getElementById('preview').classList.add('active');
            document.getElementById('form').classList.remove('active');
            document.querySelectorAll('.tab-button')[0].classList.remove('active');
            document.querySelectorAll('.tab-button')[1].classList.add('active');

            showSuccess('Dados salvos com sucesso!');
        }

        // Enviar por email via Formspree
        function sendEmail() {
            const data = JSON.parse(localStorage.getItem('declaracaoData'));
            
            if (!data) {
                showError('Nenhum dado para enviar.');
                return;
            }

            // Validar Form ID
            if (FORMSPREE_FORM_ID === 'SEU_FORM_ID_AQUI') {
                showError('❌ Erro: Form ID do Formspree não foi configurado. Por favor, atualize o arquivo com seu Form ID.');
                return;
            }

            // Mostrar indicador de carregamento
            document.getElementById('loadingIndicator').style.display = 'block';
            document.getElementById('sendEmailBtn').disabled = true;

            // Preparar dados para envio
            const formData = new FormData();
            formData.append('Nome Completo', data.nomeCompleto);
            formData.append('Nome da Empresa', data.nomeEmpresa);
            formData.append('CNPJ', data.cnpj);
            formData.append('Benefício/Desconto', data.beneficio);
            formData.append('Local', data.local);
            formData.append('Data', data.data);
            formData.append('Email do Cliente', data.email);
            formData.append('Assinatura', data.signatureImage);
            formData.append('_subject', 'Nova Declaração de Consentimento - Clube de Vantagens');

            // Enviar para Formspree
            fetch(`https://formspree.io/f/${FORMSPREE_FORM_ID}`, {
                method: 'POST',
                body: formData
            })
            .then(response => {
                if (response.ok) {
                    showSuccess('✅ Declaração enviada com sucesso para rogerio.chaves@grupolima.net.br!');
                    document.getElementById('loadingIndicator').style.display = 'none';
                    document.getElementById('sendEmailBtn').disabled = false;
                } else {
                    throw new Error('Erro ao enviar formulário');
                }
            })
            .catch(error => {
                showError('❌ Erro ao enviar: ' + error.message);
                document.getElementById('loadingIndicator').style.display = 'none';
                document.getElementById('sendEmailBtn').disabled = false;
            });
        }

        // Mostrar mensagem de sucesso
        function showSuccess(message) {
            const successMsg = document.getElementById('successMessage');
            successMsg.textContent = message;
            successMsg.classList.add('show');
            setTimeout(() => {
                successMsg.classList.remove('show');
            }, 5000);
        }

        // Mostrar mensagem de erro
        function showError(message) {
            const errorMsg = document.getElementById('errorMessage');
            errorMsg.textContent = message;
            errorMsg.classList.add('show');
            setTimeout(() => {
                errorMsg.classList.remove('show');
            }, 5000);
        }

        // Restaurar dados ao carregar a página
        window.addEventListener('load', () => {
            const savedData = localStorage.getItem('declaracaoData');
            if (savedData) {
                const data = JSON.parse(savedData);
                document.getElementById('nomeCompleto').value = data.nomeCompleto;
                document.getElementById('nomeEmpresa').value = data.nomeEmpresa;
                document.getElementById('cnpj').value = data.cnpj;
                document.getElementById('beneficio').value = data.beneficio;
                document.getElementById('local').value = data.local;
                document.getElementById('data').value = data.data;
                document.getElementById('email').value = data.email;
            }
        });
